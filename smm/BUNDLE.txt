const API_BASE = (() => {
  // 1. Пользователь указал вручную в настройках — берём оттуда
  const stored = localStorage.getItem('seo_api_base');
  if (stored) return stored.replace(/\/$/, '');
  // 2. Локальная разработка (файл:// или localhost)
  const h = location.hostname;
  if (!h || h === 'localhost' || h === '127.0.0.1') {
    return 'http://localhost:8000/api';
  }
  // 3. Продакшн на waifubot.website
  if (h.endsWith('waifubot.website')) {
    return 'https://seo.waifubot.website/api';
  }
  // 4. Общий fallback: same-origin /api
  return `${location.protocol}//${h}/api`;
})();

function saveApiBase(url) {
  localStorage.setItem('seo_api_base', url.replace(/\/$/, ''));
}

async function req(path, opts = {}) {
  const url = API_BASE.replace(/\/$/, '') + path;
  try {
    const response = await fetch(url, {
      cache: 'no-store',
      headers: {
        'Content-Type': 'application/json',
        'X-Requested-With': 'XMLHttpRequest',
        ...(opts.headers || {}),
      },
      ...opts,
    });
    if (response.status === 204) return null;
    const text = await response.text();
    if (!response.ok) {
      let msg = text;
      let detail = null;
      try { detail = JSON.parse(text)?.detail ?? null; } catch {}
      if (detail && typeof detail === 'object') {
        // Бэкенд отдаёт структуру {error, message, problems} — например,
        // когда каркас обрезан. Не превращаем его в [object Object].
        msg = detail.message || JSON.stringify(detail);
      } else if (detail) {
        msg = detail;
      }
      // Диагностика: эндпоинта нет на сервере (устаревший деплой бэкенда)
      if (response.status === 404 || response.status === 405) {
        msg = `${msg || 'Not Found'} — эндпоинт недоступен: ${url}. ` +
              `Вероятно, на сервере запущена старая версия бэкенда — обновите деплой.`;
      }
      throw new Error(`HTTP ${response.status}: ${msg}`);
    }
    try { return JSON.parse(text); } catch { return text; }
  } catch (e) {
    // TypeError = network error (CORS, DNS, SSL, fetch failed)
    if (e instanceof TypeError) {
      throw new Error(
        `Не удалось соединиться с бэкендом (${url}). ` +
        `Проверьте: сервер запущен, URL в настройках верный, SSL-сертификат валиден.`
      );
    }
    throw e;
  }
}

// ── Utils ──────────────────────────────────────────────
function escHtml(str) {
  return String(str ?? '').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
}

function fmtDate(iso) {
  if (!iso) return '—';
  const d = new Date(iso);
  return d.toLocaleDateString('ru-RU') + ' ' + d.toLocaleTimeString('ru-RU', { hour: '2-digit', minute: '2-digit' });
}

function statusLabel(status) {
  const map = {
    draft: 'Черновик', active: 'Активен', generating: 'Генерируется…',
    done: 'Готов', error: 'Ошибка',
    // pending и queued раньше оба подписывались как «В очереди», и было не понять,
    // ждёт ли ключ запуска или по нему уже создана задача.
    pending: 'В очереди', queued: 'Взят в задачу', running: 'Выполняется',
    success: 'Успех', failed: 'Ошибка', cancelled: 'Отменён',
    ready: 'Готов', approved: 'Утверждён',
  };
  return map[status] || status;
}

function statusBadgeClass(status) {
  const map = {
    draft: 'draft', active: 'active', generating: 'generating',
    done: 'done', error: 'error',
    pending: 'pending', queued: 'queued', running: 'running',
    success: 'success', failed: 'failed', cancelled: 'cancelled',
    ready: 'ready', approved: 'approved',
  };
  return map[status] || 'pending';
}

function badge(status) {
  return `<span class="badge ${statusBadgeClass(status)}">${statusLabel(status)}</span>`;
}

function getTheme() {
  try { return localStorage.getItem('seo_theme') || 'workshop'; } catch (e) { return 'workshop'; }
}

function applyTheme(name) {
  const t = THEMES[name] ? name : 'workshop';
  if (t === 'workshop') document.documentElement.removeAttribute('data-theme');
  else document.documentElement.setAttribute('data-theme', t);
  try { localStorage.setItem('seo_theme', t); } catch (e) {}
  return t;
}

// ── Паспорт модели ─────────────────────────────────
// Показывает, какую модель просили и какая реально ответила.
function modelTag(obj) {
  if (!obj) return '';
  const asked = obj.model || '';
  const got = obj.model_returned || '';
  if (!asked && !got) return '<span class="model-tag">модель не записана</span>';
  const prov = obj.provider ? escHtml(obj.provider) + ' / ' : '';
  if (got && asked && got !== asked) {
    return `<span class="model-tag warn" title="Провайдер ответил другой моделью!">⚠ просили ${prov}${escHtml(asked)} → ответила ${escHtml(got)}</span>`;
  }
  if (got) return `<span class="model-tag ok" title="Подтверждено ответом API">✓ ${prov}${escHtml(got)}</span>`;
  return `<span class="model-tag" title="Провайдер ��е вернул имя модели">${prov}${escHtml(asked)} (не подтверждено)</span>`;
}

    const container = document.getElementById('projectsContainer');

    async function loadProjects() {
      try {
        const projects = await api.listProjects();
        renderProjects(projects);
      } catch (e) {
        container.innerHTML = `<div class="alert alert-bad">❌ Ошибка загрузки: ${e.message}. Убедитесь, что бэкенд запущен.</div>`;
      }
    }

    function renderStats(projects) {
      const strip = document.getElementById('statStrip');
      if (!projects.length) { strip.style.display = 'none'; return; }
      const pages = projects.reduce((a, p) => a + (p.pages_count || 0), 0);
      const keys  = projects.reduce((a, p) => a + (p.keys_count  || 0), 0);
      const live  = projects.filter(p => p.status === 'generating').length;
      strip.style.display = '';
      strip.innerHTML = `
        <div class="stat"><div class="stat-k">Проектов</div><div class="stat-v">${projects.length}</div></div>
        <div class="stat"><div class="stat-k">Страниц готово</div><div class="stat-v">${pages}</div></div>
        <div class="stat"><div class="stat-k">Ключевых запросов</div><div class="stat-v">${keys}</div></div>
        <div class="stat"><div class="stat-k">Генерируется сейчас</div><div class="stat-v">${live}${live ? ' <small>в работе</small>' : ''}</div></div>`;
    }

    function renderProjects(projects) {
      renderStats(projects);
      if (!projects.length) {
        container.innerHTML = `
          <div class="card empty">
            <div class="empty-icon">Пустой цех</div>
            <p>Здесь появятся ваши проекты. Начните с первого — нужно только название и домен.</p>
            <button class="btn btn-primary" onclick="openCreateModal()">Создать проект</button>
          </div>`;
        return;
      }
      const cards = projects.map(p => `
        <div class="proj-wrap">
        <button class="proj-del" title="Удалить проект" aria-label="Удалить проект"
                onclick="deleteProject(event, ${p.id}, '${escHtml(p.name).replace(/'/g, "&#39;")}')">×</button>
        <a class="card proj-card" href="project.html?id=${p.id}">
          <div style="display:flex;align-items:flex-start;justify-content:space-between;gap:12px;">
            <div class="proj-name">${escHtml(p.name)}</div>
            ${badge(p.status)}
          </div>
          ${p.domain ? `<div class="proj-domain">${escHtml(p.domain)}</div>` : ''}
          ${p.description ? `<div style="color:var(--muted);font-size:13.5px;">${escHtml(p.description)}</div>` : ''}
          <div class="proj-meta">
            <span><span class="num">${p.pages_count ?? 0}</span> страниц</span>
            <span><span class="num">${p.keys_count ?? 0}</span> ключей</span>
            <span style="margin-left:auto;">${fmtDate(p.created_at)}</span>
          </div>
        </a></div>`).join('');
      container.innerHTML = `<div class="card-grid">${cards}</div>`;
    }

    // Удаление проекта целиком: клик по крестику не должен открывать карточку.
    async function deleteProject(ev, id, name) {
      ev.preventDefault();
      ev.stopPropagation();
      const ok = confirm('Удалить проект «' + name + '»?\n\nВместе с ним удалятся каркасы, ключи, все сгенерированные страницы и история чата. Отменить нельзя.');
      if (!ok) return;
      const btn = ev.currentTarget;
      if (btn) { btn.disabled = true; btn.textContent = '…'; }
      try {
        await api.deleteProject(id);
        loadProjects();
      } catch (e) {
        alert('Не удалось удалить: ' + e.message);
        if (btn) { btn.disabled = false; btn.textContent = '×'; }
      }
    }

    function openCreateModal() {
      document.getElementById('createModal').style.display = 'block';
      setTimeout(() => document.getElementById('cpName').focus(), 50);
    }
    function closeCreateModal(e) {
      if (e && e.target !== document.querySelector('#createModal .modal-overlay')) return;
      document.getElementById('createModal').style.display = 'none';
      ['cpName','cpDomain','cpDesc'].forEach(id => document.getElementById(id).value = '');
      document.getElementById('cpError').style.display = 'none';
    }
    document.getElementById('newProjectBtn').onclick = openCreateModal;

    document.getElementById('cpSubmit').onclick = async () => {
      const name   = document.getElementById('cpName').value.trim();
      const domain = document.getElementById('cpDomain').value.trim();
      const desc   = document.getElementById('cpDesc').value.trim();
      const errEl  = document.getElementById('cpError');
      if (!name) { errEl.textContent = 'Название обязательно'; errEl.style.display = 'block'; return; }
      errEl.style.display = 'none';
      const btn = document.getElementById('cpSubmit');
      btn.disabled = true; btn.textContent = 'Создание…';
      try {
        const p = await api.createProject({ name, domain, description: desc });
        closeCreateModal();
        window.location.href = `project.html?id=${p.id}`;
      } catch (e) {
        errEl.textContent = e.message;
        errEl.style.display = 'block';
      } finally {
        btn.disabled = false; btn.textContent = 'Создать проект';
      }
    };
    document.getElementById('cpName').addEventListener('keydown', e => { if (e.key === 'Enter') document.getElementById('cpSubmit').click(); });

    loadProjects();
    setInterval(() => { if (!document.hidden) loadProjects(); }, 15000);

  <script>
    const projectId = parseInt(new URLSearchParams(location.search).get('id')) || 0;
    if (!projectId) location.href = 'index.html';
    let currentProject = null, tasksInterval = null, allModels = [];

    document.querySelectorAll('.tab-btn').forEach(btn => {
      btn.addEventListener('click', () => {
        document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
        document.querySelectorAll('.tab-content').forEach(c => c.classList.remove('active'));
        btn.classList.add('active');
        document.getElementById('tab-' + btn.dataset.tab).classList.add('active');
        if (btn.dataset.tab === 'gen')    { loadTasks(); refreshGenModelHint(); if (currentProject && currentProject.status === 'generating') startTasksPolling(); } else { stopTasksPolling(); }
        if (btn.dataset.tab === 'pages')  loadPages();
        if (btn.dataset.tab === 'keys')   loadKeys();
      });
    });

    async function loadProject() {
      try {
        currentProject = await api.getProject(projectId);
        document.getElementById('projName').textContent = currentProject.name;
        document.getElementById('projStatus').innerHTML = badge(currentProject.status);
        document.getElementById('projMeta').textContent = [currentProject.domain, currentProject.description].filter(Boolean).join(' · ');
        document.getElementById('projCounts').innerHTML = `<span>📄 ${currentProject.pages_count ?? '—'} стр.</span><span>🔑 ${currentProject.keys_count ?? '—'} ключ.</span><span>⚡ ${currentProject.tasks_count ?? '—'} задач</span>`;
        document.getElementById('projLoading').style.display = 'none';
        document.getElementById('projHeader').style.display = 'block';
        loadModels();
        await Promise.all([loadChatHistory(), loadTemplate(), loadKeys(), loadTasks(), loadPages(), refreshGenModelHint()]);
        if (currentProject.status === 'generating') startTasksPolling();
      } catch (e) {
        document.getElementById('projLoading').style.display = 'none';
        document.getElementById('projError').textContent = 'Ошибка: ' + e.message;
        document.getElementById('projError').style.display = 'block';
      }
    }

    async function loadModels() {
      try { const r = await api.getModels(); allModels = r.models || []; fillModelSelects(); } catch {}
    }

    function fillModelSelects() {
      const p = currentProject?.settings || {};
      ['projCheapModel', 'projStrongModel'].forEach((id, i) => {
        const sel = document.getElementById(id);
        const current = i === 0 ? p.cheap_model?.value : p.strong_model?.value;
        const opts = ['<option value="">— из глобальных —</option>'].concat(allModels.map(m => `<option value="${escHtml(m.id)}"${m.id === current ? ' selected' : ''}>${escHtml(m.name || m.id)}</option>`));
        sel.innerHTML = opts.join('');
      });
    }

    let currentTemplate = null;
    async function loadTemplate() { /* каркас подтягивается только по кнопке «Извлечь каркас» — иначе каждое открытие проекта шлёт лишний POST */ }
    async function loadChatHistory() {
      const box = document.getElementById('chatMessages');
      try {
        const h = await api.chatHistory(projectId);
        if (!h.length) { box.innerHTML = '<div class="chat-msg system">Начните диалог.</div>'; return; }
        box.innerHTML = h.map(m => renderChatMsg(m)).join('');
        box.scrollTop = box.scrollHeight;
        // Генерация идёт на сервере, а не в этой вкладке: если ответ ещё
        // пишется, продолжаем ждать его даже после перезагрузки страницы.
        const pending = h.filter(m => m.status === 'pending').pop();
        if (pending) waitForChatAnswer(pending.id);
      } catch (e) { box.innerHTML = `<div class="chat-msg system">Ошибка: ${e.message}</div>`; }
    }

    document.getElementById('chatSend').onclick = sendChat;
    document.getElementById('chatInput').addEventListener('keydown', e => { if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); sendChat(); } });
    document.getElementById('askFrameBtn').onclick = () => { document.getElementById('chatInput').value = 'Сформировать каркас'; sendChat(); };
    document.getElementById('extractFrameBtn').onclick = async () => {
      const btn = document.getElementById('extractFrameBtn'); btn.disabled = true;
      try { const f = await api.chatExtractFrame(projectId); if (f && f.detected === false) { showFrameMsg('В последнем ответе модели нет готового каркаса. Нажмите «Сформировать каркас».', 'bad'); } else { fillFrameForm(f); } } catch (e) { showFrameMsg('Ошибка: ' + e.message, 'bad'); } finally { btn.disabled = false; }
    };

    async function sendChat() {
      const input = document.getElementById('chatInput'); const msg = input.value.trim(); if (!msg) return; input.value = '';
      const box = document.getElementById('chatMessages');
      box.innerHTML += `<div class="chat-msg user">${escHtml(msg)}</div><div class="chat-msg loading" id="chatLoading">⏳</div>`;
      box.scrollTop = box.scrollHeight; document.getElementById('chatSend').disabled = true;
      try {
        // Сервер сразу отвечает id заготовки и пишет ответ в фоне.
        const r = await api.chatSend(projectId, msg);
        await waitForChatAnswer(r.id);
      } catch (e) {
        document.getElementById('chatLoading')?.remove();
        box.innerHTML += `<div class="chat-msg assistant" style="color:var(--bad);">❌ ${escHtml(e.message)}</div>`;
      }
      finally { document.getElementById('chatSend').disabled = false; box.scrollTop = box.scrollHeight; }
    }

    // ── Ожидание фонового ответа ────────────────────────────
    // Ответ живёт в базе, а не в зависшем HTTP-запросе, поэтому смена
    // темы, перезагрузка и таймаут прокси его больше не теряют.
    let chatWaitTimer = null;

    function renderChatMsg(m) {
      if (m.status === 'pending') {
        return `<div class="chat-msg assistant" id="chatMsg${m.id}"><span class="spinner"></span> ` +
               `Модель пишет ответ… можно закрыть вкладку, он не потеряется</div>`;
      }
      if (m.status === 'error') {
        return `<div class="chat-msg assistant" id="chatMsg${m.id}" style="color:var(--bad);">` +
               `❌ Ошибка генерации: ${escHtml(m.error || 'неизвестно')}</div>`;
      }
      const tag = m.role === 'assistant' ? `<div style="margin-top:8px;">${modelTag(m)}</div>` : '';
      return `<div class="chat-msg ${m.role}" id="chatMsg${m.id}">${escHtml(m.content)}${tag}</div>`;
    }

    async function waitForChatAnswer(messageId) {
      const box = document.getElementById('chatMessages');
      if (chatWaitTimer) { clearTimeout(chatWaitTimer); chatWaitTimer = null; }
      document.getElementById('chatLoading')?.remove();
      if (!document.getElementById('chatMsg' + messageId)) {
        box.innerHTML += renderChatMsg({ id: messageId, status: 'pending', role: 'assistant' });
        box.scrollTop = box.scrollHeight;
      }
      const started = Date.now();
      const LIMIT_MS = 20 * 60 * 1000;
      let fails = 0;

      const tick = async () => {
        let m;
        try {
          m = await api.chatMessage(projectId, messageId);
        } catch (e) {
          // Сеть могла моргнуть — это не повод сразу бросать ожидание.
          // Но молчать вечно нельзя: три неудачи подряд — показываем причину.
          fails += 1;
          if (fails < 3 && Date.now() - started < LIMIT_MS) {
            chatWaitTimer = setTimeout(tick, 4000);
            return;
          }
          const bad = document.getElementById('chatMsg' + messageId);
          if (bad) bad.outerHTML = renderChatMsg({
            id: messageId, role: 'assistant', status: 'error', content: '',
            error: 'Не удалось получить ответ: ' + ((e && e.message) || e) +
                   '. Ответ модели, скорее всего, уже сохранён в базе — обновите страницу.',
          });
          return;
        }
        fails = 0;

    // ── Контроль обрезанного HTML на клиенте ──────────────────
    // Зеркало серверного html_integrity(): ловим обрыв ещё до отправки.
    const VOID_TAGS = new Set(['area','base','br','col','embed','hr','img','input','link','meta','param','source','track','wbr']);
    const OPTIONAL_CLOSE = new Set(['li','p','td','th','tr','option','dt','dd','thead','tbody','tfoot']);

    function htmlIntegrity(html) {
      const problems = [];
      const raw = String(html || '');
      if (!raw.trim()) return ['HTML пустой'];
      const lastLt = raw.lastIndexOf('<'), lastGt = raw.lastIndexOf('>');
      if (lastLt > lastGt) {
        problems.push('Разметка обрывается посреди тега: ' + raw.slice(lastLt, lastLt + 40));
        if ((raw.slice(lastLt).split('"').length - 1) % 2 === 1) problems.push('Незакрытая кавычка в атрибуте');
      }
      const opens = (raw.match(/\{\{/g) || []).length, closes = (raw.match(/\}\}/g) || []).length;
      if (opens !== closes) problems.push(`Непарные скобки плейсхолдеров: ${opens} «{{» против ${closes} «}}»`);
      if (raw.includes('```')) problems.push('Остался markdown-забор ```');
      const stack = [];
      let stray = 0;
      const re = /<\s*(\/?)\s*([a-zA-Z][a-zA-Z0-9]*)\b[^>]*?(\/?)\s*>/g;
      let m;
      while ((m = re.exec(raw)) !== null) {
        const name = m[2].toLowerCase();
        if (VOID_TAGS.has(name) || m[3]) continue;
        if (!m[1]) stack.push(name);
        else if (stack.includes(name)) { while (stack.length) { if (stack.pop() === name) break; } }
        else stray++;
      }
      const unclosed = stack.filter(t => !OPTIONAL_CLOSE.has(t));
      if (unclosed.length) problems.push(`Не закрыто тегов: ${unclosed.length} (${unclosed.slice(0, 6).map(t => '<' + t + '>').join(', ')})`);
      if (stray) problems.push(`Закрывающих тегов без пары: ${stray}`);
      const t = raw.trim();
      if (!t.endsWith('>') && !t.endsWith('}')) problems.push('Финал не похож на конец разметки');
      return problems;
    }

    function qaLetter(s) {
      if (s >= 88) return ['A', 'var(--good)'];
      if (s >= 78) return ['B', 'var(--good)'];
      if (s >= 68) return ['C', 'var(--hold)'];
      if (s >= 55) return ['D', 'var(--hold)'];
      return ['F', 'var(--alarm)'];
    }

    const PAGE_CHECKS = {};

    function checksCell(p) {
      const checks = (p && p.checks) || [];
      PAGE_CHECKS[p.id] = checks;
      const failed = checks.filter(c => !c.passed);
      const raw = p ? p.quality_score : null;
      const s = (raw === null || raw === undefined) ? null : Number(raw);

      let head;
      if (s !== null && !isNaN(s)) {
        const lt = qaLetter(s);
        head = `<div class="qa-score" style="color:${lt[1]};">${s.toFixed(0)}`
             + `<span class="qa-letter">${lt[0]}</span></div>`;

             + ` страница создана до обновления — нажми «Ещё раз»">—</div>`;

      const cnt = checks.length
        ? `<span class="qa-count ${failed.length ? 'bad' : 'ok'}">`
          + `${checks.length - failed.length}/${checks.length}</span>`
        : '';

      const nFail = failed.length
        ? `<span class="qa-failn">${failed.length} не прошло</span>` : '';

      return `<div class="qa-cell">`
        + `<div class="qa-head">${head}${cnt}</div>`
        + nFail
        + (checks.length
            ? `<button type="button" class="qa-open" data-id="${p.id}">проверки</button>`

      const payload = {
        strong_model: { value: strongSel, provider: getProvider(strongSel) },
        cheap_model:  { value: cheapSel,  provider: getProvider(cheapSel)  },
        strong_system_prompt: { value: document.getElementById('strongPrompt').value },
        cheap_system_prompt:  { value: document.getElementById('cheapPrompt').value },
        strong_max_tokens: { value: parseInt(document.getElementById('strongMaxTokens').value, 10) || 16000 },
        cheap_max_tokens:  { value: parseInt(document.getElementById('cheapMaxTokens').value, 10) || 8000 },
        gen_timeout_sec:   { value: parseInt(document.getElementById('genTimeoutSec').value, 10) || 60 },
        chat_history_limit: { value: Math.min(50, Math.max(0, parseInt(document.getElementById('chatHistoryLimit').value, 10) || 20)) },
      };

      const btn = document.getElementById('saveBtn');
      btn.disabled = true; btn.textContent = 'Сохранение…';
      try {
        await api.updateSettings(payload);
        showSaveMsg('✅ Настройки сохранены!', 'var(--ok)');
      } catch (e) {
        showSaveMsg('❌ ' + e.message, 'var(--bad)');

        btn.disabled = false; btn.textContent = 'Сохранить настройки';
      }

    async function addCustomEndpoint() {
      const name    = document.getElementById('customName').value.trim();
      const baseUrl = document.getElementById('customBaseUrl').value.trim();
      const apiKey  = document.getElementById('customApiKey').value.trim();
      if (!baseUrl) { showCustomStatus('Введите Base URL', 'bad'); return; }
      try {
        await api.addCustomEndpoint({ name, base_url: baseUrl, api_key: apiKey });
        document.getElementById('customName').value = '';
        document.getElementById('customBaseUrl').value = '';
        document.getElementById('customApiKey').value = '';
        showCustomStatus('✅ Ключ сохранён, подгружаю его модели…', 'ok');
        await init();
      } catch (e) {
        showCustomStatus('❌ ' + e.message, 'bad');
      }
    }

    async function deleteEndpoint(id) {
      if (!confirm('Удалить этот ключ? Его модели исчезнут из списков.')) return;
      try {
        const r = await api.deleteCustomEndpoint(id);
        showCustomStatus(r && r.warning ? '⚠️ ' + r.warning : '✅ Ключ удалён', r && r.warning ? 'bad' : 'ok');
        await init();
      } catch (e) {
        showCustomStatus('❌ ' + e.message, 'bad');
      }
    }

    async function testEndpoint(id, btn) {
      btn.disabled = true;
      showCustomStatus('⏳ Проверяю ключ…', '');
      try {
        const r = await api.testCustomEndpoint(id);
        showCustomStatus(r.message || (r.ok ? '✅ OK' : '⚠️ Ошибка'), r.ok ? 'ok' : 'bad');
      } catch (e) {
        showCustomStatus('❌ ' + e.message, 'bad');
      } finally {
        btn.disabled = false;
      }
    }

    // Тест идёт через бэкенд-прокси: браузерный CORS больше не мешает
    async function testCustomApi(btn) {
      const baseUrl = document.getElementById('customBaseUrl').value.trim();
      const apiKey  = document.getElementById('customApiKey').value.trim();
      if (!baseUrl) { showCustomStatus('Введите Base URL', 'bad'); return; }
      btn.disabled = true;
      showCustomStatus('⏳ Тестирование соединения через бэкенд…', '');
      try {
        const res = await api.testCustomApi(baseUrl, apiKey);
        if (res && res.ok) {
          showCustomStatus(res.message || '✅ Соединение успешно!', 'ok');
        } else {
          showCustomStatus('⚠️ ' + (res?.message || 'Неизвестный ответ сервера'), 'bad');
        }
      } catch (e) {
        showCustomStatus('❌ ' + e.message, 'bad');
      } finally {
        btn.disabled = false;
      }
    }

    function showCustomStatus(text, type) {

    }

      el.textContent = text;
      el.style.color = color;
      el.style.display = 'inline';
      setTimeout(() => el.style.display = 'none', 4000);
    }

      setTimeout(() => document.getElementById('apiBaseSaved').style.display = 'none', 3000);
      setTimeout(() => init(), 500);
    }

        </button>`).join('');
      document.querySelectorAll('#themeChoice .theme-card').forEach(el => {
        el.onclick = () => { applyTheme(el.dataset.themeId); renderThemes(); };
