                    let serverLang = String(user.language_code || 'ru')

                        const lastClaim = new Date(user.last_bonus_date);

                        const todayMidnight = Date.UTC(now.getUTCFullYear(), now.getUTCMonth(), now.getUTCDate());
                        const lastClaimMidnight = Date.UTC(lastClaim.getUTCFullYear(), lastClaim.getUTCMonth(), lastClaim.getUTCDate());
                        const diffDays = (todayMidnight - lastClaimMidnight) / 86400000;

                    renderCharacters();

        function confirmPayment(method) {
            tg.HapticFeedback.selectionChanged();
            
            const modalTitle = document.getElementById('modal-title');
            const originalText = modalTitle.innerText;
            modalTitle.innerHTML = '<i class="fa-solid fa-circle-notch fa-spin"></i> Создаем счет...';
        
            const userId = tg.initDataUnsafe?.user?.id;
            
            fetch(WEBHOOK_URL, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    action: "buy_invoice", 
                    pack_id: currentItem.id,
                    amount: method === 'stars' ? currentItem.amountStars : currentItem.amountUsd,
                    currency: method === 'stars' ? 'XTR' : 'USD', 
                    method: method,
                    chat_id: userId,
                    initData: tg.initData // ДОБАВЛЕНО
                })
            })
            .then(response => response.json())
            .then(data => {
                if (data.pay_url) {
                    tg.openLink(data.pay_url);
                    closePaymentModal();
                } 
                else if (data.ok && data.result) {
                    tg.openInvoice(data.result, (status) => {
                        if (status === 'paid') { 
                            tg.showAlert("Успешная оплата! 🎉"); 
                            closePaymentModal(); 
                        }
                    });
                } 
                else {
                    tg.showAlert("Что-то пошло не так: " + JSON.stringify(data).substring(0, 100));
                }
            })
            .catch(error => {
                tg.showAlert("Ошибка связи: " + error.message);
            })
            .finally(() => {
                modalTitle.innerText = originalText;
            });
        }

        function adminCreatePromo() {
            const code = document.getElementById('admin-promo-code').value.trim();
            const stars = document.getElementById('admin-promo-stars').value || 0;
            const gems = document.getElementById('admin-promo-gems').value || 0;
            const uses = document.getElementById('admin-promo-uses').value || 100;

            if (!code) {
                return tg.showAlert("Введи название промокода!");
            }

            // Показываем, что загрузка пошла
            tg.MainButton.showProgress();

            fetch(WEBHOOK_ADMIN, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    action: 'create_promo',
                    code: code,
                    stars: parseInt(stars),
                    gems: parseInt(gems),
                    uses_left: parseInt(uses),
                    initData: tg.initData // Обязательно передаем подпись для безопасности
                })
            })
            .then(res => res.json())
            .then(data => {
                tg.MainButton.hideProgress();
                if (data.success) {
                    showToast(`✅ Промокод ${code.toUpperCase()} создан!`, "success");
                    // Очищаем поля после успешного создания
                    document.getElementById('admin-promo-code').value = '';
                    document.getElementById('admin-promo-stars').value = '';
                    document.getElementById('admin-promo-gems').value = '';
                } else {
                    tg.showAlert("Ошибка сервера: " + (data.detail || "Неизвестная ошибка"));
                }
            })
            .catch(err => {
                tg.MainButton.hideProgress();
                tg.showAlert("Ошибка связи с сервером");
                console.error("Promo error:", err);
            });
        }

                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ chat_id: userId, initData: tg.initData })
            })

            const container = document.getElementById('my-custom-chars-container');
            const emptyState = document.getElementById('empty-mine-state');
            const fab = document.getElementById('fab-create');
            
            if (!window.customCharactersList || window.customCharactersList.length === 0) {
                container.innerHTML = '';
                emptyState.classList.remove('hidden');
                emptyState.classList.add('flex');
                if (fab) fab.classList.add('hidden'); // Нет персов - плюса нет
                return;
            }
        
            emptyState.classList.add('hidden');
            emptyState.classList.remove('flex');

            const isMineTab = document.getElementById('btn-create-mine').classList.contains('tab-active');
            const isCreatingNew = !document.getElementById('create-section-new').classList.contains('hidden');
            
            if (fab && isMineTab && !isCreatingNew) {
                fab.classList.remove('hidden'); 
            } else if (fab) {

        function voteChar(charId, isLike) {
            tg.HapticFeedback.impactOccurred('light');

                method: 'POST',
                headers: { 'Content-Type': 'application/json' },

            });

        function preloadImagesAndHideLoader() {
            // Находим все картинки, которые отрендерились в контейнере персонажей
            const images = document.querySelectorAll('#characters-container img');
            let loadedCount = 0;
            const totalImages = images.length;
        
            // Если картинок почему-то нет, просто прячем лоадер
            if (totalImages === 0) {
                hideLoadingScreen();
                return;
            }
        
            // Проверяем каждую картинку
            images.forEach(img => {
                // Если картинка уже успела загрузиться (например, из кэша)
                if (img.complete) {
                    loadedCount++;
                    if (loadedCount === totalImages) hideLoadingScreen();
                } else {
                    // Ждем события загрузки конкретной картинки
                    img.addEventListener('load', () => {
                        loadedCount++;
                        if (loadedCount === totalImages) hideLoadingScreen();
                    });
                    // Если картинка битая (ошибка 404), тоже считаем ее загруженной, 
                    // чтобы лоадер не завис навечно
                    img.addEventListener('error', () => {
                        loadedCount++;
                        if (loadedCount === totalImages) hideLoadingScreen();
                    });
                }
            });
        }

        // --- УМНОЕ ФОРМАТИРОВАНИЕ ВРЕМЕНИ ЧАТОВ ---
        function formatChatDate(timestampSec, lang) {
            const date = new Date(timestampSec * 1000);
            const now = new Date();

            // Сбрасываем часы/минуты, чтобы считать разницу именно в сутках
            const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
            const targetDay = new Date(date.getFullYear(), date.getMonth(), date.getDate());

            // Считаем разницу в днях
            const diffDays = Math.floor((today - targetDay) / (1000 * 60 * 60 * 24));

            if (diffDays === 0) {
                // Сегодня -> Часы:Минуты
                const hours = date.getHours().toString().padStart(2, '0');
                const minutes = date.getMinutes().toString().padStart(2, '0');
                return `${hours}:${minutes}`;
            } else if (diffDays < 7) {
                // В пределах недели -> День недели
                const daysRu = ['Вс', 'Пн', 'Вт', 'Ср', 'Чт', 'Пт', 'Сб'];
                const daysEn = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'];
                return lang === 'ru' ? daysRu[date.getDay()] : daysEn[date.getDay()];
            } else {
                // Больше недели -> День.Месяц
                const day = date.getDate().toString().padStart(2, '0');
                const month = (date.getMonth() + 1).toString().padStart(2, '0');
                return `${day}.${month}`;
            }
        }
